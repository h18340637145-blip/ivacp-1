```typescript
export interface GroundingItem {
  type: EvidenceAssertion["type"];
  subject: string;
  expected?: string;
  rawText: string;
  grounded: boolean;
}

export interface GroundingResult {
  ratio: number;            // grounded / 可判定断言数；无可判定断言时为 1
  total: number;            // 可判定断言数（不含 evidence_availability 占位）
  grounded: GroundingItem[];
  ungrounded: GroundingItem[];
}

export function computeGrounding(replyText: string, evidenceText: string): GroundingResult {
  const evidence = (evidenceText ?? "").trim();
  const assertions = extractEvidenceAssertions(replyText ?? "");
  const grounded: GroundingItem[] = [];
  const ungrounded: GroundingItem[] = [];

  for (const assertion of assertions) {
    // evidence_availability / reproduction 为软性占位，不计入硬接地判定
    if (assertion.type === "evidence_availability" || assertion.type === "reproduction") {
      continue;
    }
    const subjectMatched = assertion.subject
      ? includesAny(evidence, [assertion.subject])
      : false;
    const expectedMatched = assertion.expected
      ? includesAny(evidence, [assertion.expected])
      : true;
    const isGrounded = Boolean(evidence) && subjectMatched && expectedMatched;
    const item: GroundingItem = {
      type: assertion.type,
      subject: assertion.subject,
      expected: assertion.expected,
      rawText: assertion.rawText,
      grounded: isGrounded,
    };
    (isGrounded ? grounded : ungrounded).push(item);
  }

  const total = grounded.length + ungrounded.length;
  const ratio = total === 0 ? 1 : grounded.length / total;
  return { ratio, total, grounded, ungrounded };
}
```

> `includesAny` 已在本文件顶部定义（私有），`computeGrounding` 同文件可直接调用，无需 export。

```typescript
/**
 * 离线评测脚手架：回放 log/ 产物，算证据接地率(EGR) 与标注一致率。
 * 运行：npm run eval:accuracy
 * 产出：docs/golden-baseline/eval-report.{json,md}
 */
import { readFile, writeFile, readdir } from "node:fs/promises";
import { existsSync } from "node:fs";
import path from "node:path";
import { computeGrounding } from "../lib/analysis/structured-evidence-validation";

const LOG_ROOT = path.resolve("log");
const EVAL_SET_PATH = path.resolve("docs/golden-baseline/eval-set.json");
const REPORT_JSON = path.resolve("docs/golden-baseline/eval-report.json");
const REPORT_MD = path.resolve("docs/golden-baseline/eval-report.md");

interface EvalSetEntry {
  ruleGroupDir: string;
  carModelDir: string;
  issueKey: string;
  expectedSignals?: string[];
  expectedKeywords?: string[];
  notes?: string;
}

interface TicketResult {
  ticketDir: string;
  egr: number;
  total: number;
  ungrounded: string[];
  signalRecall?: number;
  keywordHitRate?: number;
}

async function readTextSafe(filePath: string): Promise<string> {
  try {
    return await readFile(filePath, "utf8");
  } catch {
    return "";
  }
}

async function buildEvidenceText(ticketDir: string): Promise<string> {
  const parts = await Promise.all([
    readTextSafe(path.join(ticketDir, "can_signal_summary.txt")),
    readTextSafe(path.join(ticketDir, "can_trace_extracted_result.txt")),
    readTextSafe(path.join(ticketDir, "signals_id.json")),
    readTextSafe(path.join(ticketDir, "extract.md")),
  ]);
  return parts.filter(Boolean).join("\n");
}

async function collectTicketDirs(): Promise<string[]> {
  const dirs: string[] = [];
  if (!existsSync(LOG_ROOT)) return dirs;
  const groups = await readdir(LOG_ROOT, { withFileTypes: true });
  for (const group of groups) {
    if (!group.isDirectory()) continue;
    const groupPath = path.join(LOG_ROOT, group.name);
    const models = await readdir(groupPath, { withFileTypes: true });
    for (const model of models) {
      if (!model.isDirectory()) continue;
      const modelPath = path.join(groupPath, model.name);
      const tickets = await readdir(modelPath, { withFileTypes: true });
      for (const ticket of tickets) {
        if (!ticket.isDirectory()) continue;
        const ticketDir = path.join(modelPath, ticket.name);
        if (existsSync(path.join(ticketDir, "final_reply.md"))) {
          dirs.push(ticketDir);
        }
      }
    }
  }
  return dirs;
}

async function loadEvalSet(): Promise<EvalSetEntry[]> {
  const raw = await readTextSafe(EVAL_SET_PATH);
  if (!raw.trim()) return [];
  try {
    const parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? (parsed as EvalSetEntry[]) : [];
  } catch {
    return [];
  }
}

function matchEvalEntry(ticketDir: string, evalSet: EvalSetEntry[]): EvalSetEntry | undefined {
  return evalSet.find((entry) => {
    const expected = path.join(LOG_ROOT, entry.ruleGroupDir, entry.carModelDir, entry.issueKey);
    return path.resolve(ticketDir) === path.resolve(expected);
  });
}

async function main(): Promise<void> {
  const ticketDirs = await collectTicketDirs();
  const evalSet = await loadEvalSet();
  const results: TicketResult[] = [];
  const skipped: string[] = [];

  for (const ticketDir of ticketDirs) {
    const replyText = await readTextSafe(path.join(ticketDir, "final_reply.md"));
    const evidenceText = await buildEvidenceText(ticketDir);
    if (!replyText.trim() || !evidenceText.trim()) {
      skipped.push(path.relative(LOG_ROOT, ticketDir));
      continue;
    }
    const grounding = computeGrounding(replyText, evidenceText);
    const result: TicketResult = {
      ticketDir: path.relative(LOG_ROOT, ticketDir),
      egr: Number(grounding.ratio.toFixed(4)),
      total: grounding.total,
      ungrounded: grounding.ungrounded.map((u) => u.rawText),
    };

    const evalEntry = matchEvalEntry(ticketDir, evalSet);
    if (evalEntry) {
      const signalsRaw = await readTextSafe(path.join(ticketDir, "signals.json"));
      const replyLower = replyText.toLowerCase();
      const expectedSignals = evalEntry.expectedSignals ?? [];
      const signalHits = expectedSignals.filter((sig) =>
        signalsRaw.toUpperCase().includes(sig.toUpperCase()),
      ).length;
      result.signalRecall = expectedSignals.length
        ? Number((signalHits / expectedSignals.length).toFixed(4))
        : undefined;
      const expectedKeywords = evalEntry.expectedKeywords ?? [];
      const keywordHits = expectedKeywords.filter((kw) =>
        replyLower.includes(kw.toLowerCase()),
      ).length;
      result.keywordHitRate = expectedKeywords.length
        ? Number((keywordHits / expectedKeywords.length).toFixed(4))
        : undefined;
    }

    results.push(result);
  }

  const egrValues = results.map((r) => r.egr);
  const meanEgr = egrValues.length
    ? Number((egrValues.reduce((a, b) => a + b, 0) / egrValues.length).toFixed(4))
    : 0;
  const labeled = results.filter((r) => r.signalRecall !== undefined);
  const meanSignalRecall = labeled.length
    ? Number((labeled.reduce((a, b) => a + (b.signalRecall ?? 0), 0) / labeled.length).toFixed(4))
    : null;
  const keyworded = results.filter((r) => r.keywordHitRate !== undefined);
  const meanKeywordHit = keyworded.length
    ? Number((keyworded.reduce((a, b) => a + (b.keywordHitRate ?? 0), 0) / keyworded.length).toFixed(4))
    : null;

  const report = {
    generatedAt: new Date().toISOString(),
    ticketCount: results.length,
    skippedCount: skipped.length,
    meanEgr,
    meanSignalRecall,
    meanKeywordHit,
    tickets: results,
    skipped,
  };

  await writeFile(REPORT_JSON, JSON.stringify(report, null, 2), "utf8");

  const mdLines: string[] = [];
  mdLines.push(`# Jira 解票准确率评测报告`);
  mdLines.push("");
  mdLines.push(`- 生成时间：${report.generatedAt}`);
  mdLines.push(`- 评测票数：${report.ticketCount}（跳过 ${report.skippedCount}）`);
  mdLines.push(`- 平均证据接地率 EGR：**${(meanEgr * 100).toFixed(1)}%**`);
  mdLines.push(`- 平均信号召回率：${meanSignalRecall === null ? "N/A" : `${(meanSignalRecall * 100).toFixed(1)}%`}`);
  mdLines.push(`- 平均结论关键判断命中率：${meanKeywordHit === null ? "N/A" : `${(meanKeywordHit * 100).toFixed(1)}%`}`);
  mdLines.push("");
  mdLines.push(`## 未接地断言热点（前 20 票）`);
  for (const r of results.filter((x) => x.ungrounded.length > 0).slice(0, 20)) {
    mdLines.push(`- \`${r.ticketDir}\`（EGR ${(r.egr * 100).toFixed(0)}%）：${r.ungrounded.join("；")}`);
  }
  await writeFile(REPORT_MD, mdLines.join("\n"), "utf8");

  console.log(`评测完成：${report.ticketCount} 票，平均 EGR ${(meanEgr * 100).toFixed(1)}%，跳过 ${report.skippedCount}`);
}

main().catch((err) => {
  console.error("eval-accuracy 失败：", err);
  process.exit(1);
});
```

```typescript
function normalizedEditDistance(a: string, b: string): number {
  const s = a.toUpperCase();
  const t = b.toUpperCase();
  const m = s.length;
  const n = t.length;
  if (m === 0) return n;
  if (n === 0) return m;
  const dp = new Array<number>(n + 1);
  for (let j = 0; j <= n; j += 1) dp[j] = j;
  for (let i = 1; i <= m; i += 1) {
    let prev = dp[0];
    dp[0] = i;
    for (let j = 1; j <= n; j += 1) {
      const tmp = dp[j];
      dp[j] = s[i - 1] === t[j - 1]
        ? prev
        : 1 + Math.min(prev, dp[j], dp[j - 1]);
      prev = tmp;
    }
  }
  return dp[n];
}

/**
 * 对精确未命中的 token 做模糊/子串补救。每 token 至多取 1 个最佳候选。
 * 命中条件：编辑距离 ≤ 2 且 ≤ 15% 较长串长度；或一方为另一方子串（缩写场景，长度 ≥ 4）。
 */
function recoverFuzzyMatches(
  configItems: ModelSignalConfigEntry[],
  unmatchedTokens: string[],
): SignalConfigMatch[] {
  const candidates: Array<{ prop: string; key: string; item: ModelSignalConfigEntry }> = [];
  for (const item of configItems) {
    const prop = String(item.prop ?? "").trim();
    if (!prop) continue;
    candidates.push({ prop, key: normalizeSignalToken(prop), item });
  }

  const recovered: SignalConfigMatch[] = [];
  for (const token of unmatchedTokens) {
    const normToken = normalizeSignalToken(token);
    if (normToken.length < 4) continue; // 太短易误配
    let best: { prop: string; item: ModelSignalConfigEntry; similarity: number } | null = null;
    for (const cand of candidates) {
      const maxLen = Math.max(normToken.length, cand.key.length);
      const distance = normalizedEditDistance(normToken, cand.key);
      const withinEdit = distance <= 2 && distance <= maxLen * 0.15;
      const isSubstring =
        cand.key.length >= 4 &&
        (cand.key.includes(normToken) || normToken.includes(cand.key));
      if (!withinEdit && !isSubstring) continue;
      const similarity = 1 - distance / maxLen;
      if (!best || similarity > best.similarity) {
        best = { prop: cand.prop, item: cand.item, similarity };
      }
    }
    if (best) {
      recovered.push({
        sourceToken: token,
        matchedBy: "fuzzy",
        matchedSignal: null,
        prop: best.prop,
        access: best.item.access,
        configItem: best.item,
        similarity: Number(best.similarity.toFixed(3)),
      });
    }
  }
  return recovered;
}
```

## 任务 4：U2b 信号自检（受限 1 次 LLM 调用）

**文件：**

- [x] **步骤 1：在 `recoverFuzzyMatches` 之后新增自检函数**

```typescript
/**
 * U2b：对仍未命中的 token 做一次受限 LLM 自检。
 * 模型只能从候选 prop 列表中选或答 null，禁止编造。失败静默跳过。
 */
async function recoverViaSelfCheck(
  configItems: ModelSignalConfigEntry[],
  unmatchedTokens: string[],
  logger: { log: (stage: string, msg: string) => void },
): Promise<SignalConfigMatch[]> {
  if (unmatchedTokens.length === 0) return [];
  const propByNorm = new Map<string, ModelSignalConfigEntry>();
  const candidateProps: string[] = [];
  for (const item of configItems) {
    const prop = String(item.prop ?? "").trim();
    if (!prop) continue;
    const norm = normalizeSignalToken(prop);
    if (!propByNorm.has(norm)) {
      propByNorm.set(norm, item);
      candidateProps.push(prop);
    }
  }
  if (candidateProps.length === 0) return [];

  // 控制 prompt 体积：候选过多时按与未命中 token 的模糊相似度预筛 top-K
  const MAX_CANDIDATES = 200;
  let shortlisted = candidateProps;
  if (candidateProps.length > MAX_CANDIDATES) {
    const scored = candidateProps.map((prop) => {
      const key = normalizeSignalToken(prop);
      const best = Math.max(
        ...unmatchedTokens.map((tok) => {
          const nt = normalizeSignalToken(tok);
          const maxLen = Math.max(nt.length, key.length) || 1;
          return 1 - normalizedEditDistance(nt, key) / maxLen;
        }),
      );
      return { prop, score: best };
    });
    scored.sort((a, b) => b.score - a.score);
    shortlisted = scored.slice(0, MAX_CANDIDATES).map((s) => s.prop);
  }

  const systemPrompt =
    "你是车载信号配置匹配助手。给定一批待匹配 token 与候选 prop 列表，" +
    "为每个 token 选出语义最接近的 prop，只能从候选列表中选；" +
    "若无任何合理对应必须返回 null，严禁编造候选列表之外的值。" +
    "只输出 JSON 数组，每项形如 {\"token\":\"\",\"matchedProp\":\"\"|null,\"reason\":\"\"}。";
  const userContent =
    `待匹配 token：\n${unmatchedTokens.join(", ")}\n\n候选 prop：\n${shortlisted.join(", ")}`;

  let raw = "";
  try {
    raw = await chat(systemPrompt, userContent, { temperature: 0.1 });
  } catch (e) {
    logger.log("AI_STEP2", `U2b 自检调用失败，跳过：${String(e).slice(0, 120)}`);
    return [];
  }

  const allowed = new Set(shortlisted.map((p) => normalizeSignalToken(p)));
  const recovered: SignalConfigMatch[] = [];
  try {
    const jsonText = raw.slice(raw.indexOf("["), raw.lastIndexOf("]") + 1);
    const parsed = JSON.parse(jsonText) as Array<{ token?: string; matchedProp?: string | null }>;
    for (const entry of parsed) {
      const token = String(entry.token ?? "").trim();
      const matchedProp = entry.matchedProp;
      if (!token || !matchedProp) continue;
      const norm = normalizeSignalToken(matchedProp);
      if (!allowed.has(norm)) continue; // 拒绝候选外的值
      const item = propByNorm.get(norm);
      if (!item) continue;
      recovered.push({
        sourceToken: token,
        matchedBy: "self_check",
        matchedSignal: null,
        prop: String(item.prop),
        access: item.access,
        configItem: item,
      });
    }
  } catch (e) {
    logger.log("AI_STEP2", `U2b 自检结果解析失败，跳过：${String(e).slice(0, 120)}`);
    return [];
  }
  return recovered;
}
```

> `chat(systemPrompt, userContent, { temperature })` 是 `lib/server/qwen.ts` 现有签名（本会话此前已加入 temperature 支持）。`logger` 形参类型与 route 内现有 logger 用法一致（仅用 `.log`）。

```typescript
const EVIDENCE_CRITIQUE_SYSTEM =
  "你是严谨的车载故障结论复核员。下面给出一份分析结论与其可用证据语料。" +
  "请删除或弱化证据语料中找不到依据的断言；无法确认的改写为“证据不足，待人工复核”；" +
  "不得新增证据之外的任何结论或数值；保持原有【现象/证据/根因/结论/置信度】结构。" +
  "只输出修订后的结论正文，不要解释你做了什么。";

/**
 * U3：Step 5 结论产出后内联证据自批判。存在未接地断言则做 1 次批判-修订调用。
 * 自愈性质，失败回退原结论，不阻断流程。
 */
async function runEvidenceSelfCritique(
  finalReply: string,
  evidenceText: string,
  logRoot: string,
  metrics: PipelineMetrics,
  logger: { log: (stage: string, msg: string) => void },
): Promise<string> {
  if (!finalReply.trim() || !evidenceText.trim()) return finalReply;

  const before = computeGrounding(finalReply, evidenceText);
  metrics.groundingBefore = Number(before.ratio.toFixed(4));
  if (before.ungrounded.length === 0) {
    await saveStepLog(logRoot, "evidence-grounding.json", JSON.stringify({
      ratioBefore: metrics.groundingBefore,
      ratioAfter: metrics.groundingBefore,
      revised: false,
      ungroundedBefore: [],
      ungroundedAfter: [],
    }, null, 2));
    metrics.groundingAfter = metrics.groundingBefore;
    return finalReply;
  }

  let revised = finalReply;
  try {
    const userContent =
      `【原结论】\n${finalReply}\n\n【可用证据语料】\n${evidenceText}\n\n` +
      `【未接地断言】\n${before.ungrounded.map((u) => u.rawText).join("；")}`;
    const out = await chat(EVIDENCE_CRITIQUE_SYSTEM, userContent, { temperature: 0.1 });
    const stripped = stripThinkTags(out).cleaned.trim();
    if (stripped) revised = stripped;
  } catch (e) {
    logger.log("AI_STEP5", `U3 证据自批判调用失败，保留原结论：${String(e).slice(0, 120)}`);
    metrics.groundingAfter = metrics.groundingBefore;
    return finalReply;
  }

  const after = computeGrounding(revised, evidenceText);
  metrics.groundingAfter = Number(after.ratio.toFixed(4));
  await saveStepLog(logRoot, "evidence-grounding.json", JSON.stringify({
    ratioBefore: metrics.groundingBefore,
    ratioAfter: metrics.groundingAfter,
    revised: true,
    ungroundedBefore: before.ungrounded.map((u) => u.rawText),
    ungroundedAfter: after.ungrounded.map((u) => u.rawText),
  }, null, 2));
  logger.log("AI_STEP5", `U3 证据自批判：接地率 ${(before.ratio * 100).toFixed(0)}% -> ${(after.ratio * 100).toFixed(0)}%`);
  return revised;
}
```

> `stripThinkTags` 是 route 内现有工具函数（Step 2-5 均用），返回 `{ cleaned, thinking }`。
